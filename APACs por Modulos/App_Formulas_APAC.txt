// ✈️ Aeroportos com dimensionamento APAC publicado.
// Tabela literal de propósito: é uma linha só, não cresce sozinha e não tem tela de cadastro.
// Quando virar duas ou três, vira lista — mesma régua do Mapa de Alocação.
colAerosApac =
Table(
    { ICAO: "SBNF"; IATA: "NVT"; Aeroporto: "NAVEGANTES" }
);;

// ⚙️ Premissas por aeroporto, da lista. A tela carrega estas em variáveis e o usuário mexe
// nelas no painel de premissas SEM gravar; gravar é um botão explícito, que cria vigência nova.
// A vigência mais recente por aeroporto é a que vale.
colParamApac =
ForAll(
    Filter(tb_apacParametros; ativo = 1) As _p;
    {
        aeroporto: _p.aeroporto;
        vigencia_inicio: _p.vigencia_inicio;
        pax_por_modulo_hora: _p.pax_por_modulo_hora;
        ocupacao_pct: _p.ocupacao_pct;
        antecedencia_min: _p.antecedencia_min;
        assentos_min: _p.assentos_min;
        apac_por_modulo: _p.apac_por_modulo;
        modulos_por_supervisor: _p.modulos_por_supervisor;
        raiosx_disponiveis: _p.raiosx_disponiveis;
        alternativa: _p.alternativa
    }
);;

// 🧱 Postos fixos — os que não dependem da malha (Acesso C, Portão Principal, Apoio).
colPostosApac =
ForAll(
    Filter(tb_apacPostosFixos; ativo = 1) As _f;
    {
        aeroporto: _f.aeroporto; posto: _f.posto; papel: _f.papel;
        hora_inicio: _f.hora_inicio; hora_fim: _f.hora_fim;
        quantidade: _f.quantidade; ordem: _f.ordem
    }
);;
