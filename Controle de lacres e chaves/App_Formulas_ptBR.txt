// ─── CREDENCIADOS DE TODOS OS AEROPORTOS ──────────────────────────────────────
// Substitui a col_colaboradores do OnStart, que trazia só o aeroporto do USUÁRIO —
// quem operava outro aeroporto abria o campo de nome e não achava ninguém.
//
// Junta as 17 tabelas da planilha (CWB em duas) numa só. Cada tela filtra pela
// coluna Aeroporto do próprio formulário:
//     Filter(nfColaboradores; Aeroporto = <aeroporto da tela>)
//
// ShowColumns deixa só as duas colunas que o app usa. Não é enfeite: as tabelas
// precisam ter exatamente o mesmo formato para caberem juntas, e as outras colunas
// (datas, por exemplo) podem vir com tipos diferentes de uma tabela para outra.
//
// A primeira leitura busca as 17 tabelas (cerca de 13 mil nomes); depois fica em
// memória durante a sessão. Tabela nova (ex.: PLU_2) entra como mais uma linha aqui.
nfColaboradores =
Ungroup(
    Table(
        {t: ShowColumns(BFH; Nome; Aeroporto)};
        {t: ShowColumns(BGX; Nome; Aeroporto)};
        {t: ShowColumns(CWB_1; Nome; Aeroporto)};
        {t: ShowColumns(CWB_2; Nome; Aeroporto)};
        {t: ShowColumns(GYN; Nome; Aeroporto)};
        {t: ShowColumns(IGU; Nome; Aeroporto)};
        {t: ShowColumns(IMP; Nome; Aeroporto)};
        {t: ShowColumns(JOI; Nome; Aeroporto)};
        {t: ShowColumns(LDB; Nome; Aeroporto)};
        {t: ShowColumns(NVT; Nome; Aeroporto)};
        {t: ShowColumns(PET; Nome; Aeroporto)};

        {t: ShowColumns(PNZ; Nome; Aeroporto)};
