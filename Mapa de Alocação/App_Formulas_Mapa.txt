// ✈️ Aeroportos com mapa de pátio publicado
colAerosMapa =
Table(
    { ICAO: "SBNF"; IATA: "NVT"; Aeroporto: "NAVEGANTES" }
);;

// 🅿️ Pátios — vêm da lista tb_patios desde 09/09/2026. A ordem manda no agrupamento visual da
// grade e a cor_hex pinta a borda esquerda da linha; aeroporto novo é linha nova, não código novo.
colPatiosTodos =
ForAll(
    Filter(tb_patios; ativo = 1) As _p;
    {
        aeroporto: _p.aeroporto; patio: _p.patio; nome_patio: _p.nome_patio; ordem: _p.ordem;
        cor_hex: _p.cor_hex; cor_texto: _p.cor_texto; sigla: _p.sigla
    }
);;

// 🛩️ Posições — vêm da lista tb_posicoes desde 09/09/2026.
// ⚠️ O campo id vem de id_posicao, NÃO do ID do SharePoint: mais de 700 lançamentos da
// tb_alocacoesMapa gravam esse número. Renumerar aponta o histórico para a posição errada,
// e nada na tela denuncia isso.
colPosicoesTodas =
ForAll(
    Filter(tb_posicoes; ativo = 1) As _p;
    {
        aeroporto: _p.aeroporto; id: _p.id_posicao; patio: _p.patio; posicao: _p.posicao; ordem: _p.ordem;
        aeronave_max: _p.aeronave_max; equip_max: _p.equip_max; classe: _p.classe;
        comp_max: _p.comp_max; env_max: _p.env_max; tipo: _p.tipo;
        contingencia: _p.contingencia; ocupa: _p.ocupa
    }
);;

// 🛡️ Posições/pátios do aeroporto ativo, com rede de segurança: se o filtro não achar nada
// (varAero em branco, aeroporto sem posição cadastrada), cai para a tabela inteira em vez de
// deixar a grade em branco. Toda tela lê estas duas, nunca colPosicoesTodas/colPatiosTodos direto.
colPosicoesAero =
With(
    { _f: Filter(colPosicoesTodas; aeroporto = varAero) };
    Filter(If(CountRows(_f) = 0; colPosicoesTodas; _f); Coalesce(tipo; "") <> "HANGAR")
);;

// Pátio que só tem hangar (HANGARES) não vira faixa na grade nem opção no filtro de pátio.
colPatiosAero =
With(
    { _f: Filter(colPatiosTodos; aeroporto = varAero) };
    Filter(If(CountRows(_f) = 0; colPatiosTodos; _f) As _pt; !IsEmpty(Filter(colPosicoesAero As _pp; _pp.patio = _pt.patio)))
);;

// 🏭 Hangares — posição cadastrada com tipo HANGAR (tela POSIÇÕES E CORES), desde 16/09/2026.
// Não é posição de pátio: sem capacidade, sem restrição, sem conflito, fora da grade. Aparece só no
// painel HANGARES do Mapa do Dia. O lançamento é RESERVA na posição do hangar — a lista não aceita outro
// tipo — e é reconhecido pela posição, não pelo tipo_registro.
colHangaresAero = Filter(colPosicoesTodas; aeroporto = varAero And tipo = "HANGAR");;
// Saída "em aberto": aeronave entra no hangar sem data de saída. Data impossível, mostrada como "em aberto".
mapSaidaEmAberto = Date(2099; 12; 31);;

// 🚪 Portões — lista tb_portoes, POR AEROPORTO, desde 17/09/2026 (tela POSIÇÕES E CORES › PORTÕES).
// A cor pinta o contorno do bloco. Antes eram fixos aqui: um segundo aeroporto herdaria os portões de Navegantes.
colPortoes =
ForAll(
    Filter(tb_portoes; aeroporto = varAero And ativo = 1) As _g;
    { portao: _g.portao; cor_hex: _g.cor_hex; ordem: _g.ordem }
);;

// 🎫 Companhias aéreas — lista tb_companhias, VALE PARA TODOS OS AEROPORTOS, desde 17/09/2026
// (tela POSIÇÕES E CORES › COMPANHIAS). Cor do preenchimento do bloco; a sigla é a gravada em cada lançamento.
colCias =
ForAll(
    Filter(tb_companhias; ativo = 1) As _c;
    { sigla: _c.sigla; nome_cia: _c.nome_cia; cor_hex: _c.cor_hex; cor_texto: _c.cor_texto; ordem: _c.ordem }
);;

// (colPrefPosicao saiu em 17/09/2026: nenhuma tela lia, e a importação usa a lista tb_prePosicao.)

// 🗂️ Tipos de registro — enumeração local, gravada como Texto na lista
colTiposReg =
Table(
    { tipo: "VOO";        rotulo: "Voo" };
    { tipo: "INTERDICAO"; rotulo: "Interdição" };
    { tipo: "RESERVA";    rotulo: "Reserva" }
);;

// 🛬 Condição do registro — em que ponto do ciclo a aeronave está.
// Veio do app "Reserva e Alocação de Pátios", que o Mapa absorve. PREVISTO é o padrão e NÃO ganha
// selo: marcar o estado normal em todo bloco viraria ruído, como já aconteceu com o tracejado.
// O selo vai como entidade numérica de HTML porque é o que o HtmlViewer da grade renderiza sem dúvida.
colCondicoes =
Table(
    { cond: "PREVISTO";   rotulo: "Previsto";   selo: "";          cor_hex: "" };
    { cond: "NO PATIO";   rotulo: "No pátio";   selo: "&#9992;";   cor_hex: hxVerde };
    { cond: "FINALIZADO"; rotulo: "Finalizado"; selo: "&#10003;";  cor_hex: hxTextoFraco }
);;

// 🅰️ Classes de posição — cor do bloco na AVIAÇÃO GERAL, onde a companhia não distingue nada.
// Toda AV. GERAL era o mesmo cinza-claro; na aviação geral quem distingue é o porte, e o porte está
// na classe da posição. Companhia continua mandando na cor de quem tem companhia de verdade.
// linha_hex (17/09/2026, pedido do Eduardo): tom do rótulo e do trilho vazio da linha, para o Pátio 3 não
// parecer tudo igual — A suave, B mais forte. Vazio = linha sem tom. Não pinta bloco.
colClasses =
Table(
    { classe: "A";           rotulo: "Classe A";    cor_hex: "#0E7490"; cor_texto: "#D6DDE0"; linha_hex: "#F2F8FA" };
    { classe: "B";           rotulo: "Classe B";    cor_hex: "#C2410C"; cor_texto: "#D6DDE0"; linha_hex: "#FBE3D4" };
    { classe: "HELICOPTERO"; rotulo: "Helicóptero"; cor_hex: "#7E22CE"; cor_texto: "#D6DDE0"; linha_hex: "" }
);;

// 🚦 O que uma regra faz. BLOQUEIO recusa a gravação; AVISO só mostra a mensagem e deixa passar.
// Nasceu do pedido do supervisor: nem toda vizinhança é proibição — algumas são só observação.
colTiposRegra =
Table(
    { tipo: "BLOQUEIO"; rotulo: "Bloqueio" };
    { tipo: "AVISO";    rotulo: "Aviso" }
);;

// 🔽 Tabelas dos seletores — coluna única `Value`, que é o que o Dropdown moderno exibe.
// Derivadas das tabelas acima: uma fonte de verdade, sem redigitar nada.
colAerosSel =
ForAll(colAerosMapa As _a; { Value: _a.Aeroporto });;

colCondicoesSel =
ForAll(colCondicoes As _c; { Value: _c.rotulo });;

colTiposRegraSel =
ForAll(colTiposRegra As _tg; { Value: _tg.rotulo });;

colTiposSel =
ForAll(colTiposReg As _t; { Value: _t.rotulo });;

colPosicoesSel =
ForAll(Sort(colPosicoesAero; ordem; SortOrder.Ascending) As _p; { Value: _p.posicao });;

colCiasSel =
With(
    { _ordenado: Sort(colCias; ordem; SortOrder.Ascending) };
    ForAll(
        Sequence(CountRows(_ordenado) + 1) As _i;
        { Value: If(_i.Value = 1; "—"; Index(_ordenado; _i.Value - 1).sigla) }
    )
);;

colPortoesSel =
With(
    { _ordenado: Sort(colPortoes; ordem; SortOrder.Ascending) };
    ForAll(
        Sequence(CountRows(_ordenado) + 1) As _i;
        { Value: If(_i.Value = 1; "—"; Index(_ordenado; _i.Value - 1).portao) }
    )
);;

colPatiosSel =
With(
    { _ordenado: Sort(colPatiosAero; ordem; SortOrder.Ascending) };
    ForAll(
        Sequence(CountRows(_ordenado) + 1) As _i;
        { Value: If(_i.Value = 1; "TODOS OS PÁTIOS"; Index(_ordenado; _i.Value - 1).nome_patio) }
    )
);;

mapDuracaoPadrao = 40;;    // minutos entre chegada e saída sugeridos no registro novo aberto pelo clique

// 📅 Época para converter (data; minuto) em minuto absoluto na linha do tempo.

// Data fixa e antiga de propósito: o valor só precisa ser o MESMO para todos os registros.
// O Brasil não tem horário de verão desde 2019, então DateDiff em dias é exato aqui.
mapDataBase      = Date(2020; 1; 1);;
