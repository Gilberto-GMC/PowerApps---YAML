// ---------- Identificação do usuário ----------
// ATENÇÃO: o 318 é o primeiro argumento do Coalesce, que devolve o primeiro
// valor não vazio. Ou seja, o app roda SEMPRE como o usuário 318; o Param e o
// LookUp nunca são alcançados. Mantive como estava — comportamento idêntico ao
// de hoje. Para voltar ao usuário real, apagar o "318;" da linha abaixo.
varIdUser = Coalesce(
    318;
    Value(Param("USUARIO"));
    LookUp(
        User;
        usrNome = User().FullName
    ).ID
);;

// Substitui o With() do OnStart: um LookUp só, reaproveitado por todas as
// fórmulas abaixo.
nfUsuario = LookUp(
    User;
    ID = varIdUser
);;

varNomeUser = nfUsuario.usrNome;;
varEmailUser = nfUsuario.Email;;
varFuncaoUser = nfUsuario.Funcao;;

varPerfilUser = nfUsuario.Perfil;;
