// 👤 Carrega informações do usuário com base no parâmetro
Set(varIdUser; Coalesce( Value(Param("USUARIO")); LookUp(User; usrNome = User().FullName).ID ));;
// Agrupa todas as variáveis relacionadas ao usuário em um só `With`
With(
    {
        usuario: LookUp(User; ID = varIdUser);
        iataUsuario: LookUp(colAeros; Aeroporto = LookUp(User; ID = varIdUser).Aeroporto).IATA
    };
    
    Set(varNomeUser; usuario.usrNome);;
    Set(varEmailUser; usuario.Email);;
    Set(varFuncaoUser; usuario.Funcao);;
    Set(varBlocoUser; usuario.Bloco);;
    Set(varAeroUser; usuario.Aeroporto);;
    Set(varPerfilUser; usuario.Perfil);;
    Set(varAutUser; usuario.Autorizacao);;
    Set(varIATAUser; iataUsuario)
);;
